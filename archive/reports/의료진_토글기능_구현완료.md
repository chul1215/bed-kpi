### 2. JavaScript 토글 기능

#### 핵심 함수: `toggleDoctorDiseases()`
```javascript
let openDoctorRow = null;

function toggleDoctorDiseases(event, doctorName) {
    event.preventDefault();
    event.stopPropagation();

    const currentRow = event.target.closest('tr');

    // 이미 열려있는 행이면 닫기
    if (openDoctorRow === currentRow) {
        const detailRow = currentRow.nextElementSibling;
        if (detailRow && detailRow.classList.contains('doctor-detail-row')) {
            detailRow.remove();
            openDoctorRow = null;
        }
        return;
    }

    // 이전에 열려있던 행 닫기
    if (openDoctorRow) {
        const prevDetailRow = openDoctorRow.nextElementSibling;
        if (prevDetailRow && prevDetailRow.classList.contains('doctor-detail-row')) {
            prevDetailRow.remove();
        }
    }

    openDoctorRow = currentRow;

    // 로딩 중 표시
    const detailRow = document.createElement('tr');
    detailRow.className = 'doctor-detail-row';
    detailRow.innerHTML = `...로딩 메시지...`;
    currentRow.after(detailRow);

    // 질환별 데이터 가져오기
    setTimeout(() => {
        const diseaseData = getDoctorDiseaseData(doctorName);
        displayDoctorDiseases(detailRow, diseaseData);
    }, 300);
}
```

#### 데이터 제공: `getDoctorDiseaseData()`
```javascript
// 의료진별 질환 데이터 (서버에서 주입)
const doctorDiseaseData = {{ doctor_disease_data|tojson|safe }};

function getDoctorDiseaseData(doctorName) {
    if (doctorDiseaseData && doctorDiseaseData[doctorName]) {
        return doctorDiseaseData[doctorName];
    }
    return [
        { diagnosis: "질환 데이터가 없습니다", ... }
    ];
}
```

#### 테이블 렌더링: `displayDoctorDiseases()`
```javascript
function displayDoctorDiseases(detailRow, diseases) {
    let diseaseTableHTML = `
        <table style="font-size: 12px; ...">
            <thead>
                <tr>
                    <th>질환명</th>
                    <th>환자수</th>
                    <th>평균재원일수 (병원 / 심평원)</th>
                    <th>LOS 갭</th>
                    <th>추가재원일수(일)</th>
                </tr>
            </thead>
            <tbody>
                ${diseases.map(disease => `
                    <tr>
                        <td>${disease.diagnosis}</td>
                        <td>${disease.patient_count}</td>
                        <td>${disease.current_los} / ${disease.target_los}</td>
                        <td>${disease.los_gap}</td>
                        <td>${disease.additional_bed_days}</td>
                    </tr>
                `).join('')}
            </tbody>
        </table>
    `;

    detailRow.innerHTML = `<td colspan="6">...</td>`;
}
```

home_html = dashboard_gen.render_home(

def render_home(
    self,
    summary_kpi: Dict[str, float],
    doctor_kpis: pd.DataFrame,
    insights: List[str],
    hospital: str = '대전',
    period: str = 'off_season',

    template = self.env.get_template('index.html')
